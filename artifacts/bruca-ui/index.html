Mesmos estados do Button. "Comprar" carrega 2s.

css base 
/* wrapper: sombra e foco (o mask cortaria) */
.tk {
  filter: drop-shadow(3px 4px 0 var(--outline));
}
.tk__btn {
  min-height: 44px;
  padding: 10px 22px;
  font: 14px/1 "Bevan";
  letter-spacing: .06em;
  text-transform: uppercase;
  background: var(--mustard);
  color: var(--on-mustard);
  border-radius: 8px 10px 7px 9px / 9px 7px 10px 8px;
  /* entalhe de 8px em cada lado */
  mask: radial-gradient(circle at 0 50%, #0000 8px, #000 0) left,
        radial-gradient(circle at 100% 50%, #0000 8px, #000 0) right;
}
.tk__btn::before {
  border: 3px solid var(--outline);
  filter: url(#sketch);
}
